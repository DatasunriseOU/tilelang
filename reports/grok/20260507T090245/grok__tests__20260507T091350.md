  ```python
  assert any("program_id" in (o.get("name") if isinstance(o, dict) else "")
  ```

  ```python
  assert any("get_program_id" in (o if isinstance(o, str) else o.get("name", "")) for o in ops)
  ```

  Also add explicit `assert "tt.get_program_id" in ops or "tt.program_id" in ops` (covers both spellings in `OP_TABLE`).

```python
def test_from_triton_kernel_vector_add_roundtrip():
    from poc.triton_frontend import from_triton_kernel
    import triton  # noqa: F401
    @triton.jit
    def triton_vec_add(A, B, Y, N, BLOCK: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N
        a = tl.load(A + offs, mask=mask)
        b = tl.load(B + offs, mask=mask)
        tl.store(Y + offs, a + b, mask=mask)

    prim = from_triton_kernel(triton_vec_add, grid=(1,), constexprs={"BLOCK": 128})
```
