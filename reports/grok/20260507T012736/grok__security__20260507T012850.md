# poc/triton_frontend/op_mapping.py:196 (replace existing map_tt_load)
def map_tt_load(op: Any, ctx: WalkerCtx) -> Any:
    """Lower tt.load → guarded BufferLoad (RFC 5.1)."""
    tir = ctx.tir()
    operands = _operands(op)
    ptr_ssa = operands[0]
    mask_ssa = operands[1] if len(operands) >= 2 else None
    other_ssa = operands[2] if len(operands) >= 3 else None

    # ptr_analysis (or MVP fallback) gives (buf, indices)
    resolved = ctx.get(ptr_ssa)
    if isinstance(resolved, tuple) and len(resolved) == 2:
        buf, indices = resolved
    else:
        buf, indices = resolved, [0]

    load_expr = tir.BufferLoad(buf, list(indices))

    if mask_ssa is not None:
        mask_expr = ctx.get(mask_ssa)
        other_expr = ctx.get(other_ssa) if other_ssa is not None else tir.const(0, _dtype_of(_results(op)[0]))
        load_expr = tir.if_then_else(mask_expr, load_expr, other_expr)

    if _results(op):
        ctx.bind(_results(op)[0], load_expr)
    return load_expr
```

### 4.2 `op_mapping.py:map_tt_dot` (full, improved)

def map_tt_dot(op: Any, ctx: WalkerCtx) -> Any:
    """Lower tt.dot → T.gemm (RFC 5.1)."""
    operands = _operands(op)
    a_ssa, b_ssa = operands[0], operands[1]
    c_ssa = operands[2] if len(operands) >= 3 else None

    a = ctx.get(a_ssa)
    b = ctx.get(b_ssa)

    attrs = _attrs(op)
    trans_a = bool(attrs.get("transpose_A", False) or attrs.get("trans_a", False))
    trans_b = bool(attrs.get("transpose_B", False) or attrs.get("trans_b", False))

    import tilelang.language as T
    if c_ssa is not None:
        c = ctx.get(c_ssa)
    else:
        # fresh accumulator (Triton fp16 → fp32 default)
        result = _results(op)[0] if _results(op) else None
        out_shape = list(_shape_of(result)) if result else []
        out_dtype = "float32" if _dtype_of(result) in {"float16", "bfloat16"} else _dtype_of(result)
        c = T.alloc_fragment(out_shape, out_dtype)

    handle = T.gemm(a, b, c, transpose_A=trans_a, transpose_B=trans_b)
    ctx.emit(handle)
    if _results(op):
        ctx.bind(_results(op)[0], c)  # in-place accumulator
    return handle
```

## 5. Killer ambiguity

**The TTIR hook point itself** (`__init__.py:275` + `_walk_mlir_module:148`).  
If we had chosen TTGIR (post-layout) we would be forced to undo `#blocked`/`#mma` encodings and re-implement layout inference — exactly what `microsoft/triton-shared` avoided and what the RFC §5.2 stance correctly rejects. The current TTIR choice (pre-layout) is the **right** one: TileLang re-derives layouts (`tilelang/transform/__init__.py:57` `LayoutInference`). Any other pivot kills portability (especially Metal).

**Recommendation**: ship exactly as-is. The scaffold is solid, security-clean, and ready for the conformance kernels.
